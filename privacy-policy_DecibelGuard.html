/** * js/app.js - 앱 메인 진입점 * 탭 전환, 측정 시작/정지, UI 업데이트 통합 관리 */ (function () { 'use strict'; // ── DOM 요소 ── const btnMeasure = document.getElementById('btn-measure'); const fabIconPlay = document.getElementById('fab-icon-play'); const fabIconStop = document.getElementById('fab-icon-stop'); const dbValue = document.getElementById('db-value'); const statMin = document.getElementById('stat-min'); const statAvg = document.getElementById('stat-avg'); const statMax = document.getElementById('stat-max'); const noiseBadge = document.getElementById('noise-level-badge'); const noiseLevelIcon = document.getElementById('noise-level-icon'); const noiseLevelText = document.getElementById('noise-level-text'); const measuringTime = document.getElementById('measuring-time'); // 탭 전환 const navBtns = document.querySelectorAll('.nav-btn'); const tabContents = document.querySelectorAll('.tab-content'); // 탭 버튼 참조 const navHistory = document.getElementById('nav-history'); const navSettings = document.getElementById('nav-settings'); // 기록/설정 탭 잠금/해제 function lockOtherTabs(lock) { [navHistory, navSettings].forEach(btn => { btn.disabled = lock; if (lock) { btn.setAttribute('title', window.i18n.t('mic.measuring')); } else { btn.removeAttribute('title'); } }); } // 현재 활성 탭 추적 (기본값: 측정 탭) let activeTab = 'measure'; // 시작 버튼 활성/비활성 처리 function updateMeasureBtn(tab) { if (tab === 'measure') { btnMeasure.disabled = false; btnMeasure.removeAttribute('title'); } else { btnMeasure.disabled = true; btnMeasure.setAttribute('title', window.i18n.t('mic.measure_only')); } } navBtns.forEach(btn => { btn.addEventListener('click', () => { const target = btn.dataset.tab; if (!target) return; // data-tab 없는 버튼(시작 버튼)은 탭 전환 스킵 activeTab = target; updateMeasureBtn(target); navBtns.forEach(b => { b.classList.toggle('nav-btn--active', b.dataset.tab === target); b.setAttribute('aria-selected', b.dataset.tab === target ? 'true' : 'false'); }); tabContents.forEach(t => { t.classList.toggle('tab-content--active', t.id === `tab-${target}`); }); // 기록 탭 진입 시 렌더링 if (target === 'history') window.HistoryManager.render(); if (target === 'settings') window.SettingsManager.syncUI(); }); }); // ── 소음 등급 업데이트 ── const LEVELS = [ { max: 40, cls: 'noise-badge--safe', icon: '🌿', key: 'badge.quiet' }, { max: 55, cls: 'noise-badge--moderate', icon: '🔊', key: 'badge.moderate' }, { max: 70, cls: 'noise-badge--caution', icon: '⚠️', key: 'badge.noisy' }, { max: 85, cls: 'noise-badge--warning', icon: '🔶', key: 'badge.warning' }, { max: 999, cls: 'noise-badge--danger', icon: '🔴', key: 'badge.danger' }, ]; let lastDb = 0; // 언어 전환 시 배지 재갱신에 사용 function updateNoiseBadge(db) { const level = LEVELS.find(l => db <= l.max) || LEVELS[LEVELS.length - 1]; noiseBadge.className = `noise-badge ${level.cls}`; noiseLevelIcon.textContent = level.icon; noiseLevelText.textContent = window.i18n.t(level.key); } // ── 측정 시작/정지 ── let isMeasuring = false; btnMeasure.addEventListener('click', async () => { // 측정 탭이 아닐 때는 동작하지 않음 if (activeTab !== 'measure') return; if (isMeasuring) { // 정지 const session = window.AudioEngine.stop(); isMeasuring = false; // 탭 잠금 해제 lockOtherTabs(false); fabIconPlay.style.display = ''; fabIconStop.style.display = 'none'; btnMeasure.classList.remove('fab--stop'); btnMeasure.classList.add('fab--start'); const measureLabel = document.getElementById('btn-measure-label'); if (measureLabel) measureLabel.textContent = window.i18n.t('nav.start'); window.TimelineRenderer.stop(); window.SpectrumRenderer.stop(); window.SpectrumRenderer.reset(); // 측정 정지 시 마지막 측정값을 유지하기 위해 리셋 코드를 삭제합니다. if (session && session.samples > 0) { window.HistoryManager.addSession(session); } } else { // 시작 const result = await window.AudioEngine.start(onAudioData); if (!result.success) { let errorKey = 'mic.error'; // 기본값 (NotAllowedError 등 권한 거부) if (result.errorName === 'NotFoundError') { errorKey = 'mic.error_notfound'; } else if (result.errorName === 'NotReadableError' || result.errorName === 'TrackStartError') { errorKey = 'mic.error_notreadable'; } window.showModal({ icon: '🎙️', title: window.i18n.t('mic.error_title'), message: window.i18n.t(errorKey), confirmText: window.i18n.t('modal.ok'), }); return; } isMeasuring = true; // 탭 잠금 lockOtherTabs(true); fabIconPlay.style.display = 'none'; fabIconStop.style.display = ''; btnMeasure.classList.remove('fab--start'); btnMeasure.classList.add('fab--stop'); const measureLabel = document.getElementById('btn-measure-label'); if (measureLabel) measureLabel.textContent = window.i18n.t('nav.stop'); window.TimelineRenderer.start(); } }); let lastTextUpdateTime = 0; // ── 오디오 데이터 수신 콜백 ── function onAudioData({ db, min, avg, max, elapsed, frequencyData, timeDomainData }) { lastDb = db; const now = Date.now(); // 텍스트 수치 및 뱃지는 사람이 읽기 쉽도록 200ms(1초에 5번) 간격으로만 업데이트 if (now - lastTextUpdateTime > 200) { // dB 표시 dbValue.textContent = db > 0 ? db.toFixed(1) : '--'; // 통계 statMin.textContent = min > 0 ? min.toFixed(1) : '--'; statAvg.textContent = avg > 0 ? avg.toFixed(1) : '--'; statMax.textContent = max > 0 ? max.toFixed(1) : '--'; // 소음 등급 if (db > 0) updateNoiseBadge(db); // 타이머 measuringTime.textContent = window.AudioEngine.formatTime(elapsed); // dB 값 색상 const color = window.GaugeRenderer.getColor(db); dbValue.style.color = db > 0 ? color : 'var(--text-primary)'; dbValue.style.textShadow = db > 0 ? `0 0 30px ${color}55` : 'none'; lastTextUpdateTime = now; } // 시각적인 게이지 바늘, 스펙트럼, 타임라인은 60fps로 부드럽게 실시간 업데이트 window.GaugeRenderer.update(db); window.SpectrumRenderer.update(frequencyData); window.TimelineRenderer.update(db); } // ── 언어 변경 시 UI 갱신 ── document.addEventListener('langchange', () => { updateNoiseBadge(lastDb); const measureLabel = document.getElementById('btn-measure-label'); if (measureLabel) { measureLabel.textContent = window.i18n.t(isMeasuring ? 'nav.stop' : 'nav.start'); } }); // ── 초기화 ── window.SettingsManager.init(); window.HistoryManager.init(); // ── URL 해시에 따른 자동 탭 전환 (예: #settings 복귀 대응) ── const hash = window.location.hash; if (hash) { const tabName = hash.replace('#', ''); const targetBtn = document.querySelector(`.nav-btn[data-tab="${tabName}"]`); if (targetBtn) { targetBtn.click(); } } console.log('[DecibelGuard] 앱 초기화 완료 ✅'); })();
